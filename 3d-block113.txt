<script>
/* DỮ LIỆU CỐ ĐỊNH & LOCALSTORAGE */
let shipStorage = JSON.parse(localStorage.getItem("shipList")) || ["288706", "S53706", "S55006", "S56906", "S57606"];
let blockStorage = JSON.parse(localStorage.getItem("blockHistory")) || ["E41P", "2E31", "H12C"];
let accessHistory = JSON.parse(localStorage.getItem("accessHistory")) || [];
let currentManagingType = null;
let currentLoadedUrl = "";

/* KHỞI TẠO VÀ PHỤC HỒI DỮ LIỆU */
window.addEventListener("DOMContentLoaded", () => {
  updateGreeting();
  getLocation();
  renderHistory();

  const lastShip = localStorage.getItem("ship");
  const lastBlock = localStorage.getItem("block");
  if (lastShip) document.getElementById("shipInput").value = lastShip;
  if (lastBlock) document.getElementById("blockInput").value = lastBlock;

  restoreActiveViewer();
});

/* DUY TRÌ TRẠNG THÁI KHI ĐỔI APP */
document.addEventListener("visibilitychange", () => {
  if (document.visibilityState === "visible") {
    restoreActiveViewer();
  }

  restoreActiveViewer();
});

function restoreActiveViewer() {
  const savedUrl = sessionStorage.getItem("current3DUrl");
  const savedShip = localStorage.getItem("ship");
  const savedBlock = localStorage.getItem("block");

  if (savedUrl) {
    const frame = document.getElementById("viewerFrame");
    if (!frame.src || frame.src === "about:blank" || frame.style.display === "none") {
      currentLoadedUrl = savedUrl;
      frame.src = savedUrl;
      frame.style.display = "block";
      document.getElementById("emptyPlaceholder").style.display = "none";
      if (savedShip && savedBlock) {
        document.getElementById("activeModelLabel").innerHTML = `Đang xem: <b>${savedShip}</b> - Block: <b>${savedBlock}</b>`;
      }
    }
  }
}

/* THỜI GIAN & ĐỊA ĐIỂM */
function updateGreeting() {
  const now = new Date();
  const hours = now.getHours();
  let greeting = "Xin chào!";
  if (hours >= 3 && hours < 10) greeting = "Chào buổi sáng 🌞";
  else if (hours >= 10 && hours < 15) greeting = "Chào buổi trưa ☀️";
  else if (hours >= 15 && hours < 18) greeting = "Chào buổi chiều 🌤️";
  else greeting = "Chào buổi tối 🌙";

  document.getElementById("greetingText").innerText = greeting;
  document.getElementById("currentDateText").innerText = `Hôm nay, ngày ${now.getDate()} tháng ${now.getMonth() + 1} năm ${now.getFullYear()}`;
}

function getLocation() {
  if (!navigator.geolocation) {
    document.getElementById("locationText").innerText = "⚠ Không hỗ trợ định vị";
    return;
  }
  navigator.geolocation.getCurrentPosition(
    (pos) => {
      fetch(`https://nominatim.openstreetmap.org/reverse?format=json&lat=${pos.coords.latitude}&lon=${pos.coords.longitude}&addressdetails=1`)
        .then(res => res.json())
        .then(data => {
          const addr = data.address;
          const village = addr.village || addr.hamlet || "";
          const ward = addr.suburb || addr.city_district || addr.town || "";
          const province = addr.state || addr.city || "";
          document.getElementById("locationText").innerText = `📍 ${[village, ward, province].filter(Boolean).join(", ")}`;
        })
        .catch(() => {
          document.getElementById("locationText").innerText = "📍 Vị trí: Đang cập nhật";
        });
    },
    () => {
      document.getElementById("locationText").innerText = "⚠ Chưa bật chia sẻ vị trí";
    }
  );
}

/* XỬ LÝ NHẬP LIỆU & GỢI Ý */
const shipInput = document.getElementById("shipInput");
const blockInput = document.getElementById("blockInput");

shipInput.addEventListener("input", function() {
  this.value = this.value.toUpperCase();
  if (this.value.length > 6) this.value = this.value.slice(0, 6);
  if (this.value.length === 6) blockInput.focus();
  renderSuggestions("ship", this.value);
});

blockInput.addEventListener("input", function() {
  this.value = this.value.toUpperCase();
  renderSuggestions("block", this.value);
});

function renderSuggestions(type, query) {
  const box = document.getElementById(type + "Suggest");
  box.innerHTML = "";
  if (!query) return;

  const list = type === "ship" ? shipStorage : blockStorage;
  const filtered = list.filter(item => item.includes(query));

  filtered.forEach(val => {
    const row = document.createElement("div");
    row.className = "suggest-row";
    row.innerHTML = `<span>${val}</span><i class="fa-solid fa-arrow-turn-down" style="font-size:11px; color:#aaa;"></i>`;
    row.onclick = () => {
      document.getElementById(type + "Input").value = val;
      box.innerHTML = "";
      if (type === "ship") blockInput.focus();
    };
    box.appendChild(row);
  });
}

document.addEventListener("click", (e) => {
  if (!e.target.closest(".input-container")) {
    document.getElementById("shipSuggest").innerHTML = "";
    document.getElementById("blockSuggest").innerHTML = "";
  }
});

/* TẢI MÔ HÌNH 3D */
function loadModel3D(customShip, customBlock) {
  const ship = (customShip || shipInput.value).trim().toUpperCase();
  const block = (customBlock || blockInput.value).trim().toUpperCase();

  const shipErr = document.getElementById("shipError");
  const blockErr = document.getElementById("blockError");

  if (ship.length !== 6) {
    shipErr.style.display = "block";
    shipInput.focus();
    setTimeout(() => shipErr.style.display = "none", 3000);
    return;
  }
  if (!block) {
    blockErr.style.display = "block";
    blockInput.focus();
    setTimeout(() => blockErr.style.display = "none", 3000);
    return;
  }

  shipInput.value = ship;
  blockInput.value = block;

  localStorage.setItem("ship", ship);
  localStorage.setItem("block", block);

  if (!blockStorage.includes(block)) {
    blockStorage.unshift(block);
    localStorage.setItem("blockHistory", JSON.stringify(blockStorage));
  }
  if (!shipStorage.includes(ship)) {
    shipStorage.unshift(ship);
    localStorage.setItem("shipList", JSON.stringify(shipStorage));
  }

  addAccessHistory(ship, block);

  const bar = document.getElementById("loadBar");
  bar.style.width = "40%";

  currentLoadedUrl = `https://mobile3d.hmd.co.kr:8087/VIZWide3D/OutfitMobile3dhttps.html?ship=${ship}&block=${block}`;
  
  sessionStorage.setItem("current3DUrl", currentLoadedUrl);

  const frame = document.getElementById("viewerFrame");
  document.getElementById("emptyPlaceholder").style.display = "none";
  frame.style.display = "block";
  frame.src = currentLoadedUrl;

  document.getElementById("activeModelLabel").innerHTML = `Đang xem: <b>${ship}</b> - Block: <b>${block}</b>`;

  frame.onload = () => {
    bar.style.width = "100%";
    setTimeout(() => bar.style.width = "0%", 300);
  };
}

/* TIỆN ÍCH XEM 3D */
function reloadViewer() {
  const frame = document.getElementById("viewerFrame");
  if (frame.src && frame.src !== "about:blank") frame.src = frame.src;
}

function openFullscreen() {
  const elem = document.getElementById("frameWrapper");
  if (elem.requestFullscreen) elem.requestFullscreen();
  else if (elem.webkitRequestFullscreen) elem.webkitRequestFullscreen();
}

function openExternalTab() {
  if (currentLoadedUrl) window.open(currentLoadedUrl, '_blank');
}

function closeViewer() {
  sessionStorage.removeItem("current3DUrl");
  currentLoadedUrl = "";

  const frame = document.getElementById("viewerFrame");
  frame.src = "about:blank";
  frame.style.display = "none";

  document.getElementById("emptyPlaceholder").style.display = "flex";
  document.getElementById("activeModelLabel").innerHTML = "Chưa chọn mô hình";
}

/* LỊCH SỬ TRUY CẬP */
function addAccessHistory(ship, block) {
  const now = new Date();
  const timeStr = `${now.getHours().toString().padStart(2, '0')}:${now.getMinutes().toString().padStart(2, '0')} - ${now.getDate()}/${now.getMonth() + 1}`;
  
  accessHistory = accessHistory.filter(i => !(i.ship === ship && i.block === block));
  accessHistory.unshift({ ship, block, time: timeStr });
  if (accessHistory.length > 10) accessHistory.pop();

  localStorage.setItem("accessHistory", JSON.stringify(accessHistory));
  renderHistory();
}

function renderHistory() {
  const listEl = document.getElementById("historyList");
  listEl.innerHTML = "";
  accessHistory.forEach(item => {
    const div = document.createElement("div");
    div.className = "history-item-row";
    div.innerHTML = `
      <span><b>${item.ship}</b> / ${item.block}</span>
      <span style="font-size:11px; color:#888;">${item.time}</span>
    `;
    div.onclick = () => loadModel3D(item.ship, item.block);
    listEl.appendChild(div);
  });
}

function toggleHistory() {
  const list = document.getElementById("historyList");
  const chevron = document.getElementById("historyChevron");
  if (list.style.display === "none") {
    list.style.display = "flex";
    chevron.className = "fa-solid fa-chevron-up";
  } else {
    list.style.display = "none";
    chevron.className = "fa-solid fa-chevron-down";
  }
}

/* MODAL QUẢN LÝ DANH SÁCH */
function openModal(type) {
  currentManagingType = type;
  document.getElementById("modalTitle").innerText = type === "ship" ? "Quản lý Danh sách Tàu" : "Quản lý Danh sách Block";
  document.getElementById("manageModal").classList.add("active");
  renderModalList();
}

function closeModal() {
  document.getElementById("manageModal").classList.remove("active");
}

function renderModalList() {
  const listEl = document.getElementById("modalItemsList");
  listEl.innerHTML = "";
  const list = currentManagingType === "ship" ? shipStorage : blockStorage;

  list.forEach((item, index) => {
    const row = document.createElement("div");
    row.className = "modal-item";
    row.innerHTML = `
      <span>${item}</span>
      <i class="fa-solid fa-trash-can" onclick="deleteModalItem(${index})"></i>
    `;
    listEl.appendChild(row);
  });
}

function addItemFromModal() {
  const input = document.getElementById("modalInputNew");
  const val = input.value.trim().toUpperCase();
  if (!val) return;

  if (currentManagingType === "ship") {
    if (!shipStorage.includes(val)) shipStorage.push(val);
    localStorage.setItem("shipList", JSON.stringify(shipStorage));
  } else {
    if (!blockStorage.includes(val)) blockStorage.push(val);
    localStorage.setItem("blockHistory", JSON.stringify(blockStorage));
  }

  input.value = "";
  renderModalList();
}

function deleteModalItem(index) {
  if (currentManagingType === "ship") {
    shipStorage.splice(index, 1);
    localStorage.setItem("shipList", JSON.stringify(shipStorage));
  } else {
    blockStorage.splice(index, 1);
    localStorage.setItem("blockHistory", JSON.stringify(blockStorage));
  }
  renderModalList();
}
