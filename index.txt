// 1. Tự động nén ảnh xuống dưới 100KB trực tiếp tại Client
function xuLyNenAnh(input) {
    if (!input.files || !input.files[0]) return;
    const tep = input.files[0];
    const reader = new FileReader();

    reader.onload = function(e) {
        const hinhAnh = new Image();
        hinhAnh.onload = function() {
            const canvas = document.createElement('canvas');
            let rong = hinhAnh.width;
            let cao = hinhAnh.height;
            const kichThuocMax = 1200;

            if (rong > kichThuocMax || cao > kichThuocMax) {
                if (rong > cao) {
                    cao = Math.round((cao * kichThuocMax) / rong);
                    rong = kichThuocMax;
                } else {
                    rong = Math.round((rong * kichThuocMax) / cao);
                    cao = kichThuocMax;
                }
            }

            canvas.width = rong;
            canvas.height = cao;
            const context = canvas.getContext('2d');
            context.drawImage(hinhAnh, 0, 0, rong, cao);

            let chatLuong = 0.85;
            let chuoiBase64 = canvas.toDataURL('image/jpeg', chatLuong);

            // Thuật toán giảm chất lượng đến khi dung lượng nhỏ hơn 100KB
            while (chuoiBase64.length * (3/4) > 100 * 1024 && chatLuong > 0.15) {
                chatLuong -= 0.1;
                chuoiBase64 = canvas.toDataURL('image/jpeg', chatLuong);
            }

            document.getElementById('anh_base64').value = chuoiBase64;
            const thongBao = document.getElementById('thongBaoNenAnh');
            thongBao.classList.remove('d-none');
            const dungLuongKb = Math.round(chuoiBase64.length * (3/4) / 1024);
            thongBao.textContent = `Ảnh đã nén thành công: ~${dungLuongKb} KB (Đạt chuẩn < 100KB)`;
        };
        hinhAnh.src = e.target.result;
    };
    reader.readAsDataURL(tep);
}

// 2. Nhận diện giọng nói tiếng Việt tự động điền form
let boNhanDien;
let dangThuAm = false;

if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
    const LopGiongNoi = window.SpeechRecognition || window.webkitSpeechRecognition;
    boNhanDien = new LopGiongNoi();
    boNhanDien.lang = 'vi-VN';
    boNhanDien.continuous = false;
    boNhanDien.interimResults = false;

    boNhanDien.onresult = function(event) {
        const cauNoi = event.results[0][0].transcript;
        phanTichGiongNoiVaoForm(cauNoi);
        dungThuAm();
    };

    boNhanDien.onerror = function() {
        dungThuAm();
    };

    boNhanDien.onend = function() {
        dungThuAm();
    };
}

function batTatGiongNoi() {
    if (!boNhanDien) {
        alert("Trình duyệt không hỗ trợ nhận diện giọng nói. Hãy dùng Google Chrome trên điện thoại hoặc máy tính có HTTPS.");
        return;
    }
    if (!dangThuAm) {
        boNhanDien.start();
        dangThuAm = true;
        document.getElementById('nutMic').className = "btn btn-danger btn-sm fw-bold";
        document.getElementById('nutMic').innerHTML = '<i class="fa-solid fa-microphone-lines fa-fade me-1"></i> Đang lắng nghe...';
        document.getElementById('trangThaiGiongNoi').innerText = "Nói theo mẫu: 'Tàu [tên], vị trí [vị trí], nội dung [nội dung]'";
    } else {
        boNhanDien.stop();
        dungThuAm();
    }
}

function dungThuAm() {
    dangThuAm = false;
    document.getElementById('nutMic').className = "btn btn-outline-primary btn-sm fw-bold";
    document.getElementById('nutMic').innerHTML = '<i class="fa-solid fa-microphone me-1"></i> Bắt đầu nói';
    document.getElementById('trangThaiGiongNoi').innerText = "Bấm mic để đọc tự động điền form";
}

function phanTichGiongNoiVaoForm(vanBan) {
    const chuThuong = vanBan.toLowerCase();
    
    // Tự động bóc tách các từ khóa
    const khopTau = chuThuong.match(/tàu\s+([^,]+?)(?=\s+vị trí|\s+nội dung|$)/i);
    const khopViTri = chuThuong.match(/vị trí\s+([^,]+?)(?=\s+nội dung|\s+tàu|$)/i);
    const khopNoiDung = chuThuong.match(/nội dung\s+(.+)$/i);

    if (khopTau) document.getElementById('ship_name').value = khopTau[1].trim();
    if (khopViTri) document.getElementById('location').value = khopViTri[1].trim();
    
    if (khopNoiDung) {
        document.getElementById('content').value = khopNoiDung[1].trim();
    } else if (!khopTau && !khopViTri) {
        // Trường hợp nói tự do không theo cấu trúc, điền toàn bộ vào nội dung
        document.getElementById('content').value = vanBan;
    }
}
