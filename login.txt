<?php
require_once 'config.php';
$thong_bao_loi = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $tai_khoan = trim($_POST['username']);
    $mat_khau = trim($_POST['password']);

    // So sánh trực tiếp chuỗi mật khẩu không qua hàm mã hóa/hash
    $stmt = $pdo->prepare("SELECT * FROM users WHERE username = ? AND password = ?");
    $stmt->execute([$tai_khoan, $mat_khau]);
    $user = $stmt->fetch();

    if ($user) {
        $_SESSION['user_id'] = $user['id'];
        $_SESSION['username'] = $user['username'];
        header("Location: index.php");
        exit;
    } else {
        $thong_bao_loi = "Tên đăng nhập hoặc mật khẩu không chính xác!";
    }
}
include 'header.php';
?>
<div class="row justify-content-center pt-5">
    <div class="col-md-4 col-sm-10">
        <div class="card shadow border-0 rounded-3">
            <div class="card-header bg-primary text-white text-center py-3">
                <h5 class="mb-0 fw-bold"><i class="fa-solid fa-lock me-2"></i>ĐĂNG NHẬP HỆ THỐNG</h5>
            </div>
            <div class="card-body p-4">
                <?php if ($thong_bao_loi): ?>
                    <div class="alert alert-danger py-2 text-center small"><?= $thong_bao_loi; ?></div>
                <?php endif; ?>
                <form method="POST">
                    <div class="mb-3">
                        <label class="form-label fw-semibold">Tên đăng nhập:</label>
                        <input type="text" name="username" class="form-control" placeholder="Nhập tài khoản" required autofocus>
                    </div>
                    <div class="mb-4">
                        <label class="form-label fw-semibold">Mật khẩu:</label>
                        <input type="password" name="password" class="form-control" placeholder="Nhập mật khẩu" required>
                    </div>
                    <button type="submit" class="btn btn-primary w-100 fw-bold py-2">
                        <i class="fa-solid fa-right-to-bracket me-1"></i> Đăng Nhập
                    </button>
                </form>
            </div>
        </div>
    </div>
</div>
<?php include 'footer.php'; ?>
